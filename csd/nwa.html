Assumptions

  • Tracking data has already been cleaned and pre-processed

    • First centre bounce starts at 0 seconds

    • 18 players from each team on the field at all times

    • Initial parsing step if this isn’t the case

  • Both data sets have errors in x and y positions and timing

    • Tracking data will have positional errors but we will use them in the final output because the relative position of the players is what we care about and we will assume the positional error is roughly to equal for all players and in the same direction

      • This won’t be the case especially with congested areas but it is the best we have

    • Tracking data is collectively continuously and automatically we will assume that each observation is actually 1 second apart

      • Just the starting point may be incorrect

    • Event data relies on an operator to click to approximate the place and time an event will occur and won’t be consistent with either

      • E.g. timestamps will lag behind when events happen quickly and then catch up when they slow down and for stoppages

Needleman-Wunch Algorithm (NWA)

Oonk et al.

  • Designed to align amino-acid sequences

  • Doesn’t rely on pre-trained models like other options (ETSY)

  • Ensures that the events happen in order and can exclude if incorrectly labelled

  • Adjustable cost function that can be altered for AFL (time and distance from the expected)

  • Compares all position of the relevant player at all frames with the expected event position and therefore could be slow for 25Hz data over multiple matches

    • Can alleviate this be breaking it up into segments

Segmentation

  • The tracking data only differentiates quarters and this may still be too large

  • Potential to train a model to recognise chain ends e.g. stoppages and goals but I don’t have the data and may introduce more errors

  • Manually annotating too labour intensive

  • One easily discernible event is the centre bounce due to the 6-6-6 rule and average approx. 4/quarter

    • Both team centroids in the centre circle

    • Minimum x < -71 and maximum x > 71 - due to players in the goal square

    • Team convex surface area consistent

hull_summary <- tracking %>%
  filter(onBench == 0) %>%
  group_by(squad_name, period_secs) %>%
  summarise(
    min_x = min(x),
    max_x = max(x),
    centroid_x = mean(x),
    centroid_y = mean(y),
    hull = list({
      i <- chull(x, y)
      tibble(x = x[i], y = y[i])
    }),
    .groups = "drop"
  ) %>%
  mutate(
    area = map_dbl(hull, ~ abs(sum(.x$x * c(.x$y[-1], .x$y[1]) - c(.x$x[-1], .x$x[1]) * .x$y) / 2))
  )

hull_summary <- left_join(hull_summary,
                          tracking %>% select(period_secs) %>% unique(),
                          by = "period_secs") 

hull_summary <- hull_summary %>%
  group_by(period_secs) %>%
  mutate(ballup_flag = all(min_x < -71 &
                             max_x > 71 &
                             between(centroid_x, -10, 10) &
                             between(centroid_y, -10, 10) &
                             between(area, 3600, 4400))) %>%
  ungroup()

ballup <- hull_summary %>%
  distinct(period_secs, ballup_flag) %>%
  arrange(period_secs) %>%
  mutate(run = cumsum(ballup_flag != lag(ballup_flag, default = !first(ballup_flag)))) %>%
  filter(ballup_flag) %>%
  group_by(run) %>%

  group_by(merge_grp) %>%

nw_align <- function(cost_matrix, gap_penalty) {
  n_events <- nrow(cost_matrix)
  n_frames <- ncol(cost_matrix)
  
  # dp[i, j] = min cost aligning events 1..i with frames 1..j
  # where frame skips at the START and END are free, but event skips cost gap_penalty
  dp <- matrix(Inf, n_events + 1, n_frames + 1)
  tb <- matrix(0L,  n_events + 1, n_frames + 1)
  
  dp[1, ] <- 0          # leading frame skips are free (event hasn't started)
  tb[1, ] <- 3L         # came from the left
  tb[1, 1] <- 0L
  
  dp[, 1] <- cumsum(c(0, rep(gap_penalty, n_events)))  # leading event skips cost
  tb[2:(n_events + 1), 1] <- 2L
  
  for (i in 2:(n_events + 1)) {
    for (j in 2:(n_frames + 1)) {
      m <- dp[i - 1, j - 1] + cost_matrix[i - 1, j - 1]   # match
      e <- dp[i - 1, j]     + gap_penalty                 # skip event
      f <- dp[i,     j - 1]                               # skip frame (FREE)
      
      best <- which.min(c(m, e, f))
      dp[i, j] <- c(m, e, f)[best]
      tb[i, j] <- best
    }
  }
  
  # End-gap-free on frames: pick best j in last row
  j <- which.min(dp[n_events + 1, ])
  i <- n_events + 1
  
  out <- vector("list", n_events)
  while (i > 1) {
    move <- tb[i, j]
    if (move == 1L) {                            # match
      out[[i - 1]] <- list(event = as.integer(i - 1), frame = as.integer(j - 1))
      i <- i - 1; j <- j - 1
    } else if (move == 2L) {                     # event skipped
      out[[i - 1]] <- list(event = as.integer(i - 1), frame = NA_integer_)
      i <- i - 1
    } else if (move == 3L) {                     # frame skipped
      j <- j - 1
    } else {
      stop("bad traceback at i=", i, " j=", j)
    }
  }
  
  tibble::tibble(
    event_idx     = vapply(out, `[[`, integer(1), "event"),
    matched_frame = vapply(out, `[[`, integer(1), "frame")
  )
}

result <- nw_align(cost_matrix, gap_penalty = 10)

classify_action <- function(desc) {
  case_when(
    desc %in% tag_actions       ~ "tag",
    desc %in% disposal_actions  ~ "disposal",
    desc %in% reception_actions ~ "reception",
    desc %in% stoppage_actions  ~ "stoppage",
    desc %in% terminal_actions  ~ "terminal",
    desc %in% contested_actions ~ "contested",
    TRUE                        ~ "unknown"
  )
}

ball <- ball %>%
  mutate(
    is_event_frame = !is.na(ev_x),
    pre_segment    = period_secs < min(synced_events$period_secs),
    
    frame_state = case_when(
      pre_segment                                          ~ "pre_segment",
      is_event_frame                                       ~ "event_moment",
      
      # Clean delivery: disposal → reception or terminal (a shot)
      prev_state == "disposal" &
        next_state %in% c("reception", "terminal")         ~ "in_flight",
      
      # Disposal didn't cleanly arrive (intercepted, spoiled, fumbled)
      prev_state == "disposal"                             ~ "in_dispute",
      
      # Carrier has the ball after a reception
      prev_state == "reception"                            ~ "in_possession",
      
      # Stoppage in progress → contested until next event
      prev_state == "stoppage"                             ~ "in_dispute",
      
      # Contested moment (spoil, dropped mark) → loose ball
      prev_state == "contested"                            ~ "in_dispute",
      
      # Out of play after a goal/behind/oob until next stoppage
      prev_state == "terminal"                             ~ "out_of_play",
      
      TRUE                                                 ~ "unknown"
    )
  )

ball <- ball %>%
  mutate(
    ball_x = case_when(
      frame_state == "event_moment"   ~ as.numeric(ev_x),
      frame_state == "in_possession"  ~ player_track_x,
      frame_state %in% c("in_flight", "in_dispute") ~
        prev_x + (period_secs - prev_secs) /
        (next_secs - prev_secs) * (next_x - prev_x),
      TRUE                            ~ NA_real_
    ),
    ball_y = case_when(
      frame_state == "event_moment"   ~ as.numeric(ev_y),
      frame_state == "in_possession"  ~ player_track_y,
      frame_state %in% c("in_flight", "in_dispute") ~
        prev_y + (period_secs - prev_secs) /
        (next_secs - prev_secs) * (next_y - prev_y),
      TRUE                            ~ NA_real_
    ),
    carrier_id = case_when(
      frame_state == "event_moment"   ~ ev_player,
      frame_state == "in_possession"  ~ prev_player,
      TRUE                            ~ NA_integer_
    )
  )
